cmake_minimum_required(VERSION 3.13)
project(cbackup VERSION 1.2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(LibArchive REQUIRED)
find_package(OpenSSL REQUIRED)

add_executable(cbackup cbackup.cpp)
if(MSVC)
  target_compile_options(cbackup PRIVATE /W4 /utf-8 /EHsc)
  target_compile_definitions(cbackup PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(cbackup PRIVATE -Wall -Wextra)
endif()
if(TARGET LibArchive::LibArchive)
  target_link_libraries(cbackup PRIVATE LibArchive::LibArchive)
else()
  target_include_directories(cbackup PRIVATE ${LibArchive_INCLUDE_DIRS})
  target_link_libraries(cbackup PRIVATE ${LibArchive_LIBRARIES})
endif()
target_link_libraries(cbackup PRIVATE OpenSSL::Crypto)
if(WIN32)
  enable_language(RC)
  target_sources(cbackup PRIVATE windows.rc)
  target_link_libraries(cbackup PRIVATE shell32 advapi32)
  target_compile_definitions(cbackup PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
endif()
# Oudere GCC (Raspberry Pi OS Buster, GCC 8) heeft std::filesystem in een aparte bibliotheek.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(cbackup PRIVATE stdc++fs)
endif()

install(TARGETS cbackup RUNTIME DESTINATION bin)
